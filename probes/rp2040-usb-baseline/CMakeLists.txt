cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(rp2040_usb_baseline C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(rp2040-usb-baseline main.c)
target_link_libraries(rp2040-usb-baseline pico_stdlib hardware_uart)
pico_enable_stdio_usb(rp2040-usb-baseline 1)
pico_enable_stdio_uart(rp2040-usb-baseline 0)
pico_add_extra_outputs(rp2040-usb-baseline)
